local function tween(o, t, p, s, d)
    local tw = TweenService:Create(o, TweenInfo.new(t, s or Enum.EasingStyle.Quint, d or Enum.EasingDirection.Out), p)
    tw:Play()
    return tw
end

local isClosing = false
local autoCloseTask
local gradientConn

local closeNotif

wrapScale.Scale = 0.85

local function show()
    tween(wrapScale, 0.6, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    tween(blur, 0.6, { Size = 18 }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

    if CONFIG.Duration > 0 then
        tween(prog, CONFIG.Duration, { Size = UDim2.fromScale(0, 1) }, Enum.EasingStyle.Linear)
        autoCloseTask = task.delay(CONFIG.Duration, function()
            closeNotif()
        end)
    end
end

closeNotif = function()
    if isClosing then return end
    isClosing = true

    pcall(function()
        if autoCloseTask then task.cancel(autoCloseTask) end
    end)
    pcall(function()
        if gradientConn then gradientConn:Disconnect() end
    end)

    pcall(function()
        tween(blur, 0.4, { Size = 0 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
        tween(wrapScale, 0.4, { Scale = 0.85 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
    end)

    task.delay(0.5, function()
        pcall(function()
            if wrap and wrap.Parent then wrap:Destroy() end
        end)
        pcall(function()
            if blur and blur.Parent then blur:Destroy() end
        end)
        pcall(function()
            if screenGui and screenGui.Parent then screenGui:Destroy() end
        end)
    end)
end

_G.closeNotifHD = closeNotif

local t = 0
local w1, w2, w3 = 0, 0, 0
gradientConn = RunService.RenderStepped:Connect(function(dt)
    if not wrap or not wrap.Parent then return end
    t = t + dt
    local target1 = math.sin(t * 0.7) * 0.4
    local target2 = math.sin(t * 0.45) * 0.25
    local target3 = math.sin(t * 1.05) * 0.5

    w1 = w1 + (target1 - w1) * math.min(dt * 4, 1)
    w2 = w2 + (target2 - w2) * math.min(dt * 3, 1)
    w3 = w3 + (target3 - w3) * math.min(dt * 5, 1)

    boxGradient.Offset = Vector2.new(w1, 0)
    innerGradient.Offset = Vector2.new(w2, 0)
    titleGradient.Offset = Vector2.new(w3, 0)
    progGradient.Offset = Vector2.new(w1 * 0.8, 0)
    imgHolderGradient.Offset = Vector2.new(w2 * 0.9, 0)
end)

end)

closeBtn.MouseButton1Click:Connect(closeNotif)

closeBtn.TouchTap:Connect(closeNotif)

task.wait(0.3)
show()
